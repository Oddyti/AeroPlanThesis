\newcommand{\inputgraduate}
{
    \coverstyle
    \inputpage{cover}

    \prevstyle
    \inputpage{previous}
    \inputpage{toc}

    % Main contents
    \bodystyle
    \inputbody{content}

    % Post part
    \poststyle
    \inputbody{post}
}

    \inputpage{proposal/cover}

    \prevstyle

    % Main contents
    \bodystyle
    \inputbody{content}

    % 开题参考文献页（独立可覆盖文件）
    \inputbody{proposal/ref}
}
